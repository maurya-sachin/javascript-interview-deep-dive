# Data shards and manifest Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the three monolithic data files (`index.json`, `search.json`, `flashcards.json`) with one small `manifest.json` plus per-topic shards loaded on demand, without changing what any page shows.

**Architecture:** `content:build` writes `manifest.json` (one overview per topic: counts, stats, no chapter or question lists) and `topics/<slug>/{index,search,flashcards}.json`. The app loads the manifest at the root route and a topic's shard through a new nested `:topic` route; Practice and Flashcards loaders return their own topic. The build keeps writing the old files until the app no longer reads them, so every commit builds and runs.

**Tech Stack:** Node 24 native TypeScript scripts, `node:test` for the two pure helpers, React 19, react-router 8.4, Vite, MiniSearch.

**Spec:** `docs/superpowers/specs/2026-10-14-isobath-redesign-design.md` → §1 Data: per-topic shards. This is step 1 of the spec's rollout; steps 2–7 get their own plans.

## Global Constraints
- R3: nothing derivable is hand-written; everything in `public/data/` is generated by `scripts/build.ts`.
- R9: npm; scripts run on Node 24 native TypeScript (`erasableSyntaxOnly`: no enums, no parameter properties; `.ts` import extensions).
- `public/data/` is gitignored; never commit it.
- Page behaviour and URLs are unchanged. No visual change.
- Never bulk-read `content/`, `app/public/data/` or `app/dist/`; check with `ls`, `du`, or one targeted file.
- Branch `v2`; one concern per commit, conventional messages.
- Question pages still load from `questions/<id>.json` (unchanged).

## Review Focus
- An unknown URL (`/nope`, `/javascript/nope`, `/javascript/how-javascript-runs/nope`) shows "not found", never a crash or blank page.
- A planned topic (`/react`) shows its "planned" message and makes no failing network request.
- A topic with content but no flashcards (`/flashcards/typescript`) says so and makes no failing request.
- Progress saved by the old build (`last` present, no `lastTitleHtml`) loads without error and just hides "Continue where you left off" until the next question is opened.
- A shard that fails to load (offline) is retried on the next navigation, not cached as a failure.

```ts
// Counts over chapters, shared by the app and by content:build (which writes them into the manifest).

import type { ChapterSummary, Difficulty, TopicStats } from './types.ts'

/** Totals for a set of chapters: what is readable now, and the difficulty mix of everything planned. */
export function chapterStats(chapters: ChapterSummary[]): TopicStats {
  const questions = chapters.flatMap((chapter) => chapter.questions)
  const withDifficulty = (level: Difficulty) => questions.filter((question) => question.difficulty === level).length
  return {
    total: questions.length,
    readable: questions.filter((question) => question.hasPage).length,
    core: questions.filter((question) => question.core).length,
    byDifficulty: { easy: withDifficulty('easy'), medium: withDifficulty('medium'), hard: withDifficulty('hard') },
  }
}
```

```ts
async function searchDocs(topic: Topic): Promise<SearchDoc[]> {
  const docs: SearchDoc[] = []
  for (const chapter of topic.chapters) {
    for (const question of chapter.questions) {
      const meta = metaOf(question)
      const written = hasPage(meta)
      docs.push({
        id: question.id,
        title: (await renderInline(meta.title)).text,
        topic: topic.title,
        chapter: chapter.title,
        tags: written ? meta.tags : [],
        tldr: written ? (await renderInline(meta.tldr)).text : '',
        hasPage: written,
      })
    }
  }
  return docs
}

async function flashcardsOf(deck: Deck): Promise<Flashcard[]> {
  return Promise.all(
    deck.cards.map(async (card) => ({
      id: `${deck.topic}/${card.id}`,
      topic: deck.topic,
      frontHtml: (await renderInline(card.front)).html,
      backHtml: (await renderInline(card.back)).html,
      question: card.question ?? null,
    })),
  )
}
```

```ts
// Loading and querying the data built by content:build: a manifest for every page, then one shard per topic.

import { useRouteLoaderData } from 'react-router'
import type { ChapterSummary, ContentManifest, Flashcard, QuestionPage as QuestionData, QuestionSummary, TopicSummary } from './types.ts'

export { chapterStats } from './stats.ts'

export const MANIFEST_ROUTE_ID = 'root'
export const TOPIC_ROUTE_ID = 'topic'

/** Reads a JSON file produced by content:build; its shape is trusted because both sides share `types.ts`. */
export async function fetchData<T>(file: string): Promise<T> {
  const response = await fetch(`${import.meta.env.BASE_URL}data/${file}`)
  if (!response.ok) throw new Error(`Could not load ${file} (${response.status}). Run npm run content:build.`)
  const data: unknown = await response.json()
  return data as T
}

let manifestPromise: Promise<ContentManifest> | undefined

/** One shared request: the root route, topic loaders and search all need the manifest. */
export function loadManifest(): Promise<ContentManifest> {
  manifestPromise ??= fetchData<ContentManifest>('manifest.json').catch((error: unknown) => {
    manifestPromise = undefined
    throw error
  })
  return manifestPromise
}

const topicPromises = new Map<string, Promise<TopicSummary>>()

/** A topic's chapters and questions, or `null` when the manifest has no such topic. */
export async function loadTopic(slug: string | undefined): Promise<TopicSummary | null> {
  const { topics } = await loadManifest()
  if (slug === undefined || !topics.some((topic) => topic.slug === slug)) return null
  let shard = topicPromises.get(slug)
  if (!shard) {
    shard = fetchData<TopicSummary>(`topics/${slug}/index.json`).catch((error: unknown) => {
      topicPromises.delete(slug) // allow a retry on the next navigation
      throw error
    })
    topicPromises.set(slug, shard)
  }
  return shard
}

/** The question's content, or `null` when it has no page yet (planned) or doesn't exist. */
export async function loadQuestion(id: string): Promise<QuestionData | null> {
  const topic = await loadTopic(id.split('/')[0])
  const summary = topic?.chapters.flatMap((chapter) => chapter.questions).find((question) => question.id === id)
  return summary?.hasPage ? fetchData<QuestionData>(`questions/${id}.json`) : null
}

/** A chapter's published questions in reading order, with the topic they belong to. */
export async function loadPractice(topicSlug: string | undefined, chapterSlug: string | undefined) {
  const topic = await loadTopic(topicSlug)
  const chapter = topic?.chapters.find((candidate) => candidate.slug === chapterSlug)
  const readable = chapter?.questions.filter((question) => question.hasPage) ?? []
  const questions = await Promise.all(readable.map((question) => fetchData<QuestionData>(`questions/${question.id}.json`)))
  return { topic, questions }
}

/** One topic's flashcards, with the topic. Topics without a deck have no file to fetch. */
export async function loadFlashcards(topicSlug: string | undefined) {
  const topic = await loadTopic(topicSlug)
  const cards = topic && topic.flashcardCount > 0 ? await fetchData<Flashcard[]>(`topics/${topic.slug}/flashcards.json`) : []
  return { topic, cards }
}

export function useManifest(): ContentManifest {
  // Route loader data is untyped in data-router mode; MANIFEST_ROUTE_ID's loader returns loadManifest().
  return useRouteLoaderData(MANIFEST_ROUTE_ID) as ContentManifest
}

/** The topic of the current `:topic` route, or `null` when the URL names no topic. */
export function useTopic(): TopicSummary | null {
  return useRouteLoaderData(TOPIC_ROUTE_ID) as TopicSummary | null
}
```
