# Brand (isobath identity) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give the app its isobath identity: name, contour mark and favicon, a header lockup, generated contour-map covers per topic, depth-scale colour tokens, and the CC BY-NC-SA licence notice.

**Architecture:** One pure module (`src/lib/contours.ts`) turns a seed into nested closed rings (value field → marching squares → chained rings). A `TopicCover` component draws them; a `Mark` component and a static favicon carry the logo. Colours reuse the existing depth ramp, split into four named stops. No new dependencies.

**Tech Stack:** TypeScript, React 19, Tailwind 4, `node:test` for the pure module.

**Spec:** `docs/superpowers/specs/2026-10-14-isobath-redesign-design.md` → §2 Brand. Layout reference: `docs/superpowers/specs/mockups/page-layouts.html`.

## Global Constraints
- Name shown as "isobath, by Sachin Maurya"; tagline "engineering interviews, in depth".
- Fonts Geist, Geist Mono, Newsreader (SIL OFL); accent `#2F4FD8` light / `#9DAEF7` dark; prose column 720px.
- Mark = nested irregular contours; the drawn mark is a **draft** a human designer will redraw. Originals only: nothing copied from other brands.
- Covers are original by construction: generated from the topic slug and data, no hand drawing, no AI image.
- Existing look must not regress: `--depth-from` / `--depth-to` keep their values.
- Never bulk-read `content/`, `app/public/data/`, `app/dist/`.
- R9: Node 24 native TypeScript (`erasableSyntaxOnly`, `.ts` import extensions in scripts/tests).
- Branch `v2`; one concern per commit.

## Review Focus
- Every cover must be valid for any slug and any chapter count, including a planned topic with 0 chapters and a topic with 30: closed rings, inside the frame, never empty.
- Dark mode: the mark, favicon and covers stay visible against the dark background.
- The cover must not change between renders or page loads (same slug, same art).
- A topic with only one or two chapters still gets a recognisable map (minimum 3 levels).
- Page titles and the browser tab say isobath everywhere; no "Deep Dive" left in app strings.

```ts
// Contour-map rings generated from a seed: the cover art for a topic, original by construction.
// A smooth height field (a few hills) is sampled on a grid; marching squares finds the lines where
// it crosses evenly spaced heights, and the segments are chained into closed rings.

export type Point = readonly [x: number, y: number]

/** One closed contour line. `level` 0 is the outermost (shallowest) ring. No repeated end point. */
export interface Ring {
  level: number
  points: Point[]
}

export interface ContourOptions {
  seed: string
  /** Number of nested levels, at least 1. */
  levels: number
  /** 0 to 1: steeper terrain packs the rings closer together. Default 0. */
  steepness?: number
  /** Side of the square the rings are drawn in. Default 100. */
  size?: number
}

/** Grid cells per side; the field has GRID + 1 samples per side. */
const GRID = 64

type Edge = 'top' | 'right' | 'bottom' | 'left'

/** For each 4-bit corner pattern (top-left 8, top-right 4, bottom-right 2, bottom-left 1), the edges each segment joins. */
const SEGMENTS: Record<number, [Edge, Edge][]> = {
  1: [['left', 'bottom']],
  2: [['bottom', 'right']],
  3: [['left', 'right']],
  4: [['top', 'right']],
  6: [['top', 'bottom']],
  7: [['top', 'left']],
  8: [['top', 'left']],
  9: [['top', 'bottom']],
  11: [['top', 'right']],
  12: [['left', 'right']],
  13: [['bottom', 'right']],
  14: [['left', 'bottom']],
}

/** Saddle cells (opposite corners inside) join differently depending on whether their centre is inside. */
const SADDLE: Record<number, { centreInside: [Edge, Edge][]; centreOutside: [Edge, Edge][] }> = {
  5: {
    centreInside: [['top', 'left'], ['bottom', 'right']],
    centreOutside: [['top', 'right'], ['left', 'bottom']],
  },
  10: {
    centreInside: [['top', 'right'], ['left', 'bottom']],
    centreOutside: [['top', 'left'], ['bottom', 'right']],
  },
}

function hashSeed(seed: string): number {
  let hash = 2166136261
  for (let index = 0; index < seed.length; index++) {
    hash ^= seed.charCodeAt(index)
    hash = Math.imul(hash, 16777619)
  }
  return hash >>> 0
}

/** mulberry32: a small seeded generator returning numbers in [0, 1). */
function seededRandom(seed: string): () => number {
  let state = hashSeed(seed)
  return () => {
    state = (state + 0x6d2b79f5) | 0
    let mixed = Math.imul(state ^ (state >>> 15), 1 | state)
    mixed = (mixed + Math.imul(mixed ^ (mixed >>> 7), 61 | mixed)) ^ mixed
    return ((mixed ^ (mixed >>> 14)) >>> 0) / 4294967296
  }
}

/** Two or three Gaussian hills kept away from the frame, so the field is lowest along the border. */
function heightField(random: () => number, steepness: number): number[][] {
  const hills = Array.from({ length: 2 + Math.floor(random() * 2) }, () => ({
    x: 0.3 + random() * 0.4,
    y: 0.3 + random() * 0.4,
    height: 0.6 + random() * 0.4,
    spread: (0.12 + random() * 0.08) * (1 - 0.35 * steepness),
  }))
  return Array.from({ length: GRID + 1 }, (_, row) =>
    Array.from({ length: GRID + 1 }, (_, column) => {
      const u = column / GRID
      const v = row / GRID
      return hills.reduce((sum, hill) => sum + hill.height * Math.exp(-((u - hill.x) ** 2 + (v - hill.y) ** 2) / (2 * hill.spread ** 2)), 0)
    }),
  )
}

/** The border samples are the lowest part of the field; heights above them give rings that never touch the frame. */
function borderMaximum(field: number[][]): number {
  const last = GRID
  const border = field.flatMap((row, index) => (index === 0 || index === last ? row : [row[0], row[last]]))
  return Math.max(...border)
}

function ringsAtHeight(field: number[][], height: number, level: number, size: number): Ring[] {
  const points = new Map<string, Point>()
  const neighbours = new Map<string, string[]>()

  /** A grid edge is named by the sample it starts at and its direction, so the two cells sharing it agree on its key. */
  function edgePoint(row: number, column: number, edge: Edge): string {
    const horizontal = edge === 'top' || edge === 'bottom'
    const edgeRow = edge === 'bottom' ? row + 1 : row
    const edgeColumn = edge === 'right' ? column + 1 : column
    const key = `${horizontal ? 'h' : 'v'}${edgeRow},${edgeColumn}`
    if (!points.has(key)) {
      const start = field[edgeRow][edgeColumn]
      const end = horizontal ? field[edgeRow][edgeColumn + 1] : field[edgeRow + 1][edgeColumn]
      const along = (height - start) / (end - start)
      const x = horizontal ? edgeColumn + along : edgeColumn
      const y = horizontal ? edgeRow : edgeRow + along
      points.set(key, [(x / GRID) * size, (y / GRID) * size])
    }
    return key
  }

  function link(first: string, second: string) {
    neighbours.set(first, [...(neighbours.get(first) ?? []), second])
    neighbours.set(second, [...(neighbours.get(second) ?? []), first])
  }

  for (let row = 0; row < GRID; row++) {
    for (let column = 0; column < GRID; column++) {
      const topLeft = field[row][column]
      const topRight = field[row][column + 1]
      const bottomRight = field[row + 1][column + 1]
      const bottomLeft = field[row + 1][column]
      const pattern = Number(topLeft > height) * 8 + Number(topRight > height) * 4 + Number(bottomRight > height) * 2 + Number(bottomLeft > height)
      const saddle = SADDLE[pattern]
      const centreInside = (topLeft + topRight + bottomRight + bottomLeft) / 4 > height
      const segments = saddle ? (centreInside ? saddle.centreInside : saddle.centreOutside) : (SEGMENTS[pattern] ?? [])
      for (const [from, to] of segments) link(edgePoint(row, column, from), edgePoint(row, column, to))
    }
  }

  const rings: Ring[] = []
  const visited = new Set<string>()
  for (const start of neighbours.keys()) {
    if (visited.has(start)) continue
    const ringPoints: Point[] = []
    let previous: string | undefined
    let current = start
    do {
      visited.add(current)
      ringPoints.push(points.get(current)!)
      const [first, second] = neighbours.get(current)!
      const next = first !== previous ? first : second
      previous = current
      current = next
    } while (current !== start)
    rings.push({ level, points: ringPoints })
  }
  return rings
}

export function contourRings({ seed, levels, steepness = 0, size = 100 }: ContourOptions): Ring[] {
  const field = heightField(seededRandom(seed), steepness)
  const floor = borderMaximum(field)
  const peak = Math.max(...field.flat())
  return Array.from({ length: levels }, (_, level) => ringsAtHeight(field, floor + ((peak - floor) * (level + 0.5)) / levels, level, size)).flat()
}

export function ringPath(points: readonly Point[]): string {
  return points.map(([x, y], index) => `${index === 0 ? 'M' : 'L'}${x.toFixed(1)} ${y.toFixed(1)}`).join('') + 'Z'
}
```

```tsx
import { contourRings, ringPath } from '@/lib/contours'
import { cn } from '@/lib/utils'

// Literal class names so Tailwind finds them; ring level 0 is shallow, the last level is deep.
const DEPTH_STROKE = ['stroke-depth-1', 'stroke-depth-2', 'stroke-depth-3', 'stroke-depth-4']
const MIN_LEVELS = 3
const MAX_LEVELS = 9

interface TopicCoverProps {
  slug: string
  chapterCount: number
  /** Share of the topic's questions rated hard, 0 to 1: harder topics get steeper, more crowded rings. */
  hardShare: number
  className?: string
}

/** A topic's cover: a contour map drawn from its slug, one level per chapter. Decorative. */
export function TopicCover({ slug, chapterCount, hardShare, className }: TopicCoverProps) {
  const levels = Math.min(Math.max(chapterCount, MIN_LEVELS), MAX_LEVELS)
  const rings = contourRings({ seed: slug, levels, steepness: hardShare })
  return (
    <svg viewBox="0 0 100 100" fill="none" strokeWidth="0.8" strokeLinejoin="round" aria-hidden="true" className={cn('overflow-hidden', className)}>
      {rings.map((ring, index) => (
        <path
          key={index}
          d={ringPath(ring.points)}
          className={cn(DEPTH_STROKE[Math.floor((ring.level * DEPTH_STROKE.length) / levels)], ring.level === levels - 1 && 'fill-depth-4/20')}
        />
      ))}
    </svg>
  )
}
```
